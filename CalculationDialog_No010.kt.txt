@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun CalculationDialog(
    sheet: CountingSheet,
    current: GroupCalculation?,
    done: (CalcDraft?) -> Unit
) {
    if (sheet.groups.isEmpty()) {
        done(null)
        return
    }

    var left by remember { mutableStateOf(current?.leftGroupId ?: sheet.groups.first().id) }
    var right by remember { mutableStateOf(current?.rightGroupId ?: sheet.groups.first().id) }
    var operator by remember { mutableStateOf(current?.operator ?: CalculationOperator.ADD) }
    var name by remember { mutableStateOf(current?.name ?: "") }

    Dialog(onDismissRequest = { done(null) }) {
        Surface(
            modifier = Modifier.fillMaxWidth().heightIn(max = 680.dp),
            shape = ArmyristPanelShape,
            color = ArmyristColors.WorkSurface,
            contentColor = ArmyristColors.PrimaryText,
            border = BorderStroke(1.dp, ArmyristColors.Border)
        ) {
            Column(Modifier.fillMaxWidth()) {
                Column(
                    Modifier.fillMaxWidth()
                        .background(ArmyristColors.RaisedSurface)
                        .padding(horizontal = 18.dp, vertical = 14.dp)
                ) {
                    Text(
                        if (current == null) "계산 추가" else "계산 편집",
                        style = MaterialTheme.typography.titleLarge,
                        fontWeight = FontWeight.Bold
                    )
                    Text(
                        "COUNTING · CALCULATION",
                        style = MaterialTheme.typography.labelSmall,
                        color = ArmyristColors.SecondaryText
                    )
                }

                HorizontalDivider(color = ArmyristColors.Divider)

                Column(
                    modifier = Modifier
                        .weight(1f, fill = false)
                        .verticalScroll(rememberScrollState())
                        .padding(horizontal = 18.dp, vertical = 12.dp),
                    verticalArrangement = Arrangement.spacedBy(12.dp)
                ) {
                    ArmyristSectionLabel("왼쪽 그룹")
                    CalculationGroupChoices(
                        groups = sheet.groups.sortedBy { it.order },
                        selectedId = left,
                        onSelect = { left = it }
                    )

                    HorizontalDivider(color = ArmyristColors.Divider)

                    ArmyristSectionLabel("연산")
                    FlowRow(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.spacedBy(8.dp),
                        verticalArrangement = Arrangement.spacedBy(8.dp)
                    ) {
                        ArmyristChoiceChip(
                            text = "+",
                            selected = operator == CalculationOperator.ADD,
                            onClick = { operator = CalculationOperator.ADD }
                        )
                        ArmyristChoiceChip(
                            text = "−",
                            selected = operator == CalculationOperator.SUBTRACT,
                            onClick = { operator = CalculationOperator.SUBTRACT }
                        )
                    }

                    HorizontalDivider(color = ArmyristColors.Divider)

                    ArmyristSectionLabel("오른쪽 그룹")
                    CalculationGroupChoices(
                        groups = sheet.groups.sortedBy { it.order },
                        selectedId = right,
                        onSelect = { right = it }
                    )

                    OutlinedTextField(
                        value = name,
                        onValueChange = { name = it },
                        label = { Text("계산명 (선택)") },
                        singleLine = true,
                        modifier = Modifier.fillMaxWidth(),
                        colors = OutlinedTextFieldDefaults.colors(
                            focusedContainerColor = ArmyristColors.InputSurface,
                            unfocusedContainerColor = ArmyristColors.InputSurface,
                            focusedBorderColor = ArmyristColors.PrimaryControl,
                            unfocusedBorderColor = ArmyristColors.Border
                        )
                    )
                }

                HorizontalDivider(color = ArmyristColors.Divider)

                Row(
                    modifier = Modifier.fillMaxWidth()
                        .background(ArmyristColors.RaisedSurface)
                        .padding(horizontal = 12.dp, vertical = 10.dp),
                    horizontalArrangement = Arrangement.End,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    OutlinedButton(
                        onClick = { done(null) },
                        shape = ArmyristPanelShape,
                        border = BorderStroke(1.dp, ArmyristColors.Border),
                        colors = ButtonDefaults.outlinedButtonColors(
                            containerColor = ArmyristColors.WorkSurface,
                            contentColor = ArmyristColors.PrimaryText
                        )
                    ) { Text("취소") }

                    Spacer(Modifier.width(8.dp))

                    Button(
                        onClick = {
                            done(CalcDraft(left, operator, right, name.trim()))
                        },
                        shape = ArmyristPanelShape,
                        colors = ButtonDefaults.buttonColors(
                            containerColor = ArmyristColors.PrimaryControl,
                            contentColor = ArmyristColors.OnDark
                        )
                    ) { Text("확인", fontWeight = FontWeight.Bold) }
                }
            }
        }
    }
}

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun CalculationGroupChoices(
    groups: List<CountingGroup>,
    selectedId: String,
    onSelect: (String) -> Unit
) {
    FlowRow(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.spacedBy(7.dp),
        verticalArrangement = Arrangement.spacedBy(7.dp)
    ) {
        groups.forEach { group ->
            ArmyristChoiceChip(
                text = group.name,
                selected = selectedId == group.id,
                onClick = { onSelect(group.id) }
            )
        }
    }
}

@Composable
private fun ArmyristChoiceChip(
    text: String,
    selected: Boolean,
    onClick: () -> Unit
) {
    Surface(
        onClick = onClick,
        shape = ArmyristPanelShape,
        color = if (selected) ArmyristColors.PrimaryControl else ArmyristColors.RaisedSurface,
        contentColor = if (selected) ArmyristColors.OnDark else ArmyristColors.PrimaryText,
        border = BorderStroke(
            if (selected) 2.dp else 1.dp,
            if (selected) ArmyristColors.Accent else ArmyristColors.Border
        )
    ) {
        Text(
            text = text,
            modifier = Modifier.padding(horizontal = 12.dp, vertical = 9.dp),
            fontWeight = if (selected) FontWeight.Bold else FontWeight.Medium,
            maxLines = 1
        )
    }
}
